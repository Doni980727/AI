-- Backend: Installation av Python --
Installera via hemsidan, klicka i PATH saken.
cd backend
Första gång: python -m venv venv
sedan: pip install fastapi uvicorn mistralai
pip install requests

-- Frontend: React install -- 
cd frontend
npm install

-- För att starta backend -- 
$env:MISTRAL_API_KEY = "API-Key"
$env:STABILITY_API_KEY = "API-Key"
uvicorn main:app --reload

-- För att starta frontend --
npm start
